import pandas as pd

peek = pd.read_csv(CSV_PATH, nrows=2000, low_memory=False)
print('Total columns in file:', len(peek.columns))

LABEL_COL = 'Attack_label'
TYPE_COL  = 'Attack_type' if 'Attack_type' in peek.columns else None

# identifier columns: numeric at times, meaningless on another network
ID_LIKE = {'frame.time', 'ip.src_host', 'ip.dst_host',
           'arp.src.proto_ipv4', 'arp.dst.proto_ipv4',
           'http.file_data', 'http.request.full_uri',
           'http.request.version', 'icmp.transmit_timestamp',
           'tcp.options', 'tcp.payload', 'mqtt.msg',
           'dns.qry.name', 'http.referer', 'mqtt.topic'}

numeric_cols = [col for col in peek.columns
                if col not in (LABEL_COL, TYPE_COL)
                and col not in ID_LIKE
                and pd.api.types.is_numeric_dtype(peek[col])]

keep = numeric_cols + [c for c in (LABEL_COL, TYPE_COL) if c]

print(f'Numeric features to load : {len(numeric_cols)}')
print(f'Columns skipped entirely : {len(peek.columns) - len(keep)}')

del peek; gc.collect()

08 · التحميل

الكود — تحميل البيانات

t0 = time.time()

dtypes = {col: np.float32 for col in numeric_cols}
dtypes[LABEL_COL] = np.int8

read_kwargs = dict(usecols=keep, dtype=dtypes, low_memory=False)

if SAMPLE_ROWS is None:
    df = pd.read_csv(CSV_PATH, **read_kwargs)
else:
    # the file is ordered, normal first -- a head read returns one class only
    total = sum(1 for _ in open(CSV_PATH, encoding='utf-8', errors='ignore')) - 1
    frac  = SAMPLE_ROWS / total
    parts = [ch.sample(frac=frac, random_state=SEED)
             for ch in pd.read_csv(CSV_PATH, chunksize=200_000, **read_kwargs)]
    df = pd.concat(parts, ignore_index=True)
    del parts

gc.collect()
print(f'Loaded {df.shape[0]:,} rows x {df.shape[1]} columns in {time.time()-t0:.0f}s')
print(f'DataFrame memory: {df.memory_usage(deep=True).sum()/1e9:.2f} GB')
✓ تابع الرقم ده

السطر الأخير بيطبع استهلاك الذاكرة الفعلي. المفروض يطلع حوالي 0.3 – 0.5 جيجا. لو طلع أكتر من 3 جيجا، يبقى في عمود نصي اتحمّل بالغلط.

09 · التنظيف والتقسيم

الكود — تنظيف وتقسيم وتطبيع

# clean
X = df[numeric_cols].replace([np.inf, -np.inf], np.nan).fillna(0)
const_cols = [c for c in X.columns if X[c].nunique() <= 1]
X = X.drop(columns=const_cols)
print(f'Final feature count: {X.shape[1]}')
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler

y = df[LABEL_COL].values
attack_types = df.loc[y == 1, TYPE_COL].values if TYPE_COL else None

X_tr, X_te_norm = train_test_split(X[y == 0], test_size=TEST_SPLIT,
                                       random_state=SEED)
X_attack = X[y == 1]

scaler = MinMaxScaler()
Xtr_s  = scaler.fit_transform(X_tr).astype(np.float32)
Xten_s = scaler.transform(X_te_norm).astype(np.float32)
Xatk_s = scaler.transform(X_attack).astype(np.float32)

n_feat = Xtr_s.shape[1]

# free memory -- only the arrays are needed from here on
del X, X_tr, X_te_norm, X_attack, df
gc.collect()

print(f'Attack range: {Xatk_s.min():.2f} -> {Xatk_s.max():.2f}')
⚠ غلطتين قاتلين — خد بالك

1. الـ scaler بيتعلّم من التدريب بس (fit_transform). بيانات الاختبار transform بس. لو عملت fit عليها يبقى data leakage والنتيجة مالهاش قيمة.

2. ما تقصّش القيم بـ np.clip(X, 0, 1). قيم الهجوم بتطلع برّه مدى 0→1 — وده بالظبط اللي بيخليها تتكشف. لما جرّبتها، الكشف وقع من 99% لـ 3%.

✓ علامة إنك ماشي صح

آخر سطر لازم يطبع مدى الهجوم أكبر من 1 — مثلاً 0.00 -> 6.00. لو طلع 0.00 -> 1.00 بالظبط، يبقى في clipping في مكان ما.

10 · الموديل

الكود — بناء وتدريب الموديل

import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense
from tensorflow.keras.optimizers import Adam

tf.random.set_seed(SEED)

h1, h2 = max(32, n_feat // 2), max(16, n_feat // 4)

inp = Input(shape=(n_feat,))
e   = Dense(h1, activation='relu')(inp)
e   = Dense(h2, activation='relu')(e)
d   = Dense(h1, activation='relu')(e)
out = Dense(n_feat, activation='sigmoid')(d)

ae = Model(inp, out)
ae.compile(optimizer=Adam(1e-3), loss='mse')
ae.summary()
from tensorflow.keras.callbacks import EarlyStopping

hist = ae.fit(
    Xtr_s, Xtr_s,            # input == target
    epochs=EPOCHS,
    batch_size=BATCH_SIZE,
    validation_split=0.1,
    shuffle=True,
    callbacks=[EarlyStopping(patience=3, restore_best_weights=True,
                             monitor='val_loss')],
    verbose=1
)
◆ إيه اللي اتغيّر عن نسخة Colab
  • batch_size=1024 بدل 256 — معالجك أقوى فيستحمل دفعات أكبر وده أسرع
  • EarlyStopping — يوقف لوحده لما التحسّن يقف، فما تستناش على الفاضي
★ المتوقّع

كل epoch هتاخد دقيقة ونص لتلاتة على البيانات الكاملة. الإجمالي 25 – 50 دقيقة. سيب الجهاز شغال.

11 · الحساب

الكود — حساب الخطأ على دفعات

لو حسبنا الخطأ لكل الصفوف مرة واحدة، هيتعمل مصفوفة ضخمة في الذاكرة. فبنقسّمها.

def recon_error(model, X, batch=20_000):
    """Per-row mean squared error, computed batch by batch."""
    out = np.empty(len(X), dtype=np.float32)
    for i in range(0, len(X), batch):
        chunk = X[i:i+batch]
        R = model.predict(chunk, batch_size=4096, verbose=0)
        out[i:i+batch] = np.mean(np.square(chunk - R), axis=1)
    return out

err_train = recon_error(ae, Xtr_s)
err_norm  = recon_error(ae, Xten_s)
err_atk   = recon_error(ae, Xatk_s)

threshold = float(np.percentile(err_train, THRESHOLD_PCT))
print(f'Threshold = {threshold:.6f}')
print(f'Ratio: {err_atk.mean()/err_norm.mean():.1f}x')
◆ الـ Ratio ده مؤشر سريع

بيقولك متوسط خطأ الهجوم أكبر من الطبيعي بكام ضعف. كل ما الرقم كبر، كل ما الفصل بينهم أوضح.

عشان كده بنقص العيّنة عند 50 ألف. والـ Autoencoder بيفضل متقيّم على البيانات الكاملة.

from sklearn.ensemble import IsolationForest
from sklearn.svm import OneClassSVM

rng = np.random.default_rng(SEED)

def cap(A, n):
    if len(A) <= n:
        return A
    return A[rng.choice(len(A), n, replace=False)]

fit_sample  = cap(Xtr_s,  BASELINE_CAP)
norm_sample = cap(Xten_s, BASELINE_CAP)
atk_sample  = cap(Xatk_s, BASELINE_CAP)

iso = IsolationForest(n_estimators=100, contamination=0.05,
                      random_state=SEED, n_jobs=-1).fit(fit_sample)
svm = OneClassSVM(kernel='rbf', nu=0.05, gamma='scale').fit(fit_sample[:10_000])
★ وضّح ده في العرض

15 · اختبر نفسك

أسئلة سريعة

1. ليه بنقرا 2000 صف الأول قبل التحميل الكامل؟
الأعمدة النصية بتاخد ذاكرة كتير. لما نعرف مين فيهم رقمي، بنمرّر usecols فالنصوص عمرها ما بتدخل الذاكرة أصلاً.
2. إيه اللي بيحصل لو عملت np.clip(Xatk_s, 0, 1)؟
قيم الهجوم بتطلع برّه مدى 0→1 عشان هي فعلاً شاذة. ده هو اللي بيخلي الموديل يكشفها. لما جرّبنا، الكشف وقع من 99% لـ 3%.
3. ليه الـ One-Class SVM بناخده على عيّنة؟
خوارزمية O(n²). الـ Autoencoder بالعكس — خطي، فبيتقيّم على البيانات الكاملة عادي.
4. ليه مش بنستخدم الـ RTX 3050؟
الموديل ده صغير جداً فالفرق دقيقة واحدة. الكارت هيفرق فعلاً لما توصل للـ LSTM والـ GAT — وساعتها WSL2 أو PyTorch هيحلوا المشكلة.
